\begin{frame}{Schemat quadcoptera -- jak to działa?}
    Aby maszyna mogła latać, potrzebuje czterech wirników, które generują ciąg (siłę nośną).
    \vspace{0.5cm}
    
    \begin{center}
    \begin{tikzpicture}[x=(-150:1cm), y=(0:1cm), z=(90:1cm), scale=1.8]
        
        % Środek ciężkości (CoG)
        \coordinate (O) at (0,0,0);
        \filldraw[black] (O) circle (1.5pt) node[above right, xshift=2mm, font=\scriptsize] {Środek Masy (CoG)};

        % Ramiona drona (Konfiguracja X)
        \draw[line width=1.5pt, gray!80, line cap=round] (-1,-1,0) -- (1,1,0);
        \draw[line width=1.5pt, gray!80, line cap=round] (-1,1,0) -- (1,-1,0);
        
        % Silniki i Śmigła (Rysowanie płaskich okręgów w płaszczyźnie XY)
        \begin{scope}[canvas is xy plane at z=0]
            \foreach \x/\y/\n/\dir in {-1/-1/1/CW, -1/1/2/CCW, 1/1/3/CW, 1/-1/4/CCW} {
                \filldraw[darkgray] (\x,\y) circle (0.15); % Silnik
                \draw[thick, dashed] (\x,\y) circle (0.5); % Zasięg śmigła
                \node[below, fill=white, inner sep=1pt, yshift=-3mm, font=\tiny] at (\x,\y) {$M_\n$ (\dir)};
            }
        \end{scope}

        % Osie Body Frame (Wydłużone)
        \draw[->, >=stealth, line width=1.2pt, red] (O) -- (2.5,0,0) node[below left, font=\footnotesize] {$X_B$ (Przód)};
        \draw[->, >=stealth, line width=1.2pt, green!60!black] (O) -- (0,2.5,0) node[above right, font=\footnotesize] {$Y_B$ (Prawa strona)};
        \draw[->, >=stealth, line width=1.2pt, blue] (O) -- (0,0,-2.5) node[below, font=\footnotesize] {$Z_B$ (Dół)};

        % --- Kąty Eulera z użyciem płaszczyzn 3D ---
        
        % Roll (Obrót wokół osi X_B -> Leży w płaszczyźnie YZ)
        \begin{scope}[canvas is yz plane at x=2.0]
            \draw[->, >=stealth, red, thick] (0.6, 0) arc (0:270:0.6);
        \end{scope}

        % Pitch (Obrót wokół osi Y_B -> Leży w płaszczyźnie XZ)
        \begin{scope}[canvas is xz plane at y=2.0]
            \draw[->, >=stealth, green!60!black, thick] (0.6, 0) arc (0:-270:0.6);
        \end{scope}

        % Yaw (Obrót wokół osi Z_B -> Leży w płaszczyźnie XY)
        \begin{scope}[canvas is xy plane at z=-2.0]
            \draw[->, >=stealth, blue, thick] (0.6, 0) arc (0:-270:0.6);
        \end{scope}

        % Etykiety kątów (ustawione pod osiami za pomocą współrzędnych 3D)
        \node[red, font=\footnotesize, fill=white, inner sep=1pt] at (2.0, 0, -0.8) {Przechylenie / Roll ($\phi$)};
        \node[green!60!black, font=\footnotesize, fill=white, inner sep=1pt] at (0, 2.0, -0.8) {Pochylenie / Pitch ($\theta$)};
        \node[blue, font=\footnotesize, fill=white, inner sep=1pt] at (0, 2.0, -3.0) {Odchylenie / Yaw ($\psi$)};

    \end{tikzpicture}
    \end{center}

\begin{frame}{Czym jest autonomia?}
    Autonomia w robotyce oznacza, że maszyna sama podejmuje decyzje i wie:
    \begin{itemize}
        \item Gdzie się znajduje?
        \item Gdzie ma polecieć?
        \item Jak tam dotrzeć, omijając przeszkody?
    \end{itemize}
    \vspace{0.5cm}
    \textbf{Odpowiedzią na te pytania jest matematyka.} Pojęcia z lekcji matematyki i fizyki stają się potężnymi narzędziami, z których korzystają współczesne technologie.
\end{frame}
