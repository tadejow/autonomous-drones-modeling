        \filldraw[black] (O) circle (1.5pt) node[above right, xshift=2mm, font=\scriptsize] {Środek Masy (CoG)};

        % Ramiona drona (Konfiguracja X)
        \draw[line width=1.5pt, gray!80, line cap=round] (-1,-1,0) -- (1,1,0);
        \draw[line width=1.5pt, gray!80, line cap=round] (-1,1,0) -- (1,-1,0);

\begin{frame}{Czym jest autonomia?}
    Autonomia w robotyce oznacza, że maszyna sama podejmuje decyzje i wie:
    \begin{itemize}
        \item Gdzie się znajduje?
        \item Gdzie ma polecieć?
        \item Jak tam dotrzeć, omijając przeszkody?
    \end{itemize}
    \vspace{0.5cm}
    \textbf{Odpowiedzią na te pytania jest matematyka.} Pojęcia z lekcji matematyki i fizyki stają się potężnymi narzędziami, z których korzystają współczesne technologie.
\end{frame}

\begin{frame}{Architektura Systemu -- Stacja naziemna a Dron}
    Za lotem autonomicznym stoi uporządkowany podział ról. System sterowania składa się z dwóch głównych komputerów:
    
    \vspace{0.5cm}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{1. Stacja Bazowa (Ground Control Station)}
            \begin{itemize}
                \item Np. programy QGroundControl, Mission Planner.
                \item Odpowiada za planowanie misji (punkty trasy), wyświetlanie mapy i monitorowanie stanu maszyny przez operatora siedzącego bezpiecznie na ziemi.
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{2. Kontroler Lotu (Autopilot)}
            \begin{itemize}
                \item W naszym projekcie nie ma fizycznego układu elektronicznego (jak Pixhawk).
                \item Wykorzystujemy symulator \textbf{SITL (Software-In-The-Loop)}.
                \item System \textbf{ArduPilot} działa jako lokalny proces na komputerze. Rozwiązuje na żywo równania matematyczne i fizyczne (symulując m.in. żyroskop i GPS), zastępując prawdziwego drona.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Język maszyn -- Protokół MAVLink}
    Aby Stacja Bazowa i Kontroler Lotu mogły ze sobą współpracować (często na odległość kilometrów za pomocą radia), muszą mówić w tym samym języku. Do tego służy lekki protokół \textbf{MAVLink}.
    
    \vspace{0.2cm}
    \begin{center}
    \begin{tikzpicture}[scale=0.8, transform shape,
        box/.style={draw, rounded corners, fill=blue!10, text width=2.8cm, align=center, minimum height=1.2cm},
        arrow/.style={->, thick, >=stealth}]
        
        \node[box, fill=green!20] (gcs) at (0, 0) {\textbf{Stacja Bazowa (GCS)}};
        \node[box, fill=orange!20] (mavlink) at (5.5, 0) {\textbf{Protokół MAVLink} \\ \footnotesize Telemetria, Komendy};
        \node[box, fill=gray!20] (drone) at (11, 0) {\textbf{Symulator (SITL)} \\ \footnotesize ArduPilot};
        
        \draw[arrow, bend left=20] (gcs.north east) to node[above, font=\scriptsize, align=center]{SET\_MODE (np. GUIDED)\\Wysyłanie punktów trasy} (mavlink.north west);
        \draw[arrow, bend left=20] (mavlink.north east) to node[above, font=\scriptsize, align=center]{Rozkazy dla maszyny} (drone.north west);
        
        \draw[arrow, bend left=20] (drone.south west) to node[below, font=\scriptsize, align=center]{Pozycja GPS, Kąty Eulera,\\Stan baterii} (mavlink.south east);
        \draw[arrow, bend left=20] (mavlink.south west) to node[below, font=\scriptsize, align=center]{HEARTBEAT\\(Puls maszyny)} (gcs.south east);
    \end{tikzpicture}
    \end{center}
    \vspace{0.2cm}
    Kluczowym komunikatem jest \textbf{HEARTBEAT} (bicie serca) – sygnał wysyłany co sekundę. Jeśli GCS przestanie go odbierać, wie, że utracono łączność radiową i dron automatycznie zainicjuje powrót (RTL).
\end{frame}

\begin{frame}{Zanim dron poleci -- Pre-Arm Checks}
    Bezpieczeństwo to podstawa. Zanim silniki zostaną "uzbrojone" (komenda ARM) i zaczną się kręcić, kontroler lotu wykonuje serię rygorystycznych testów fizycznych i matematycznych. Jeśli choć jeden zawiedzie, dron nie pozwoli na start.
    
    \vspace{0.3cm}
    \begin{center}
    \begin{tikzpicture}[scale=0.9]
        \node[draw, thick, fill=yellow!10, text width=11cm, align=left, inner sep=0.4cm, rounded corners] (checks) at (0,0) {
            \textbf{Filtry bezpieczeństwa (Pre-Arm Checks):}\\[0.3cm]
            \textcolor{green!60!black}{\textbf{\checkmark} GPS Lock (3D Fix)} -- Czy złapaliśmy sygnał z co najmniej 6 satelitów i precyzję trójwymiarową? Bez tego dron nie wie gdzie jest.\\[0.1cm]
            \textcolor{green!60!black}{\textbf{\checkmark} IMU Calibration} -- Czy akcelerometry i żyroskopy zgadzają się co do tego, gdzie znajduje się środek Ziemi ("dół")?\\[0.1cm]
            \textcolor{green!60!black}{\textbf{\checkmark} Compass Variance} -- Czy pole magnetyczne nie jest zaburzone (np. przez bliskość transformatora lub metalowego dachu)?\\[0.1cm]
            \textcolor{red}{\textbf{\texttimes} Battery Failsafe} -- Czy napięcie baterii wystarczy na bezpieczny lot i powrót?
        };
    \end{tikzpicture}
    \end{center}
    \end{frame}
