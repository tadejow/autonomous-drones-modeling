\section{Misje autonomiczne}

\begin{frame}{Skanowanie obszaru i optymalizacja tras}
    Ratownik szukający zaginionej osoby nie lata losowo. Wyznacza obszar do skanowania.
    Aby zminimalizować czas przelotu między punktami, system rozwiązuje Problem Komiwojażera (TSP - Traveling Salesperson Problem).
    
    \vspace{0.2cm}
    \begin{columns}
        \column{0.4\textwidth}
        \begin{center}
        \begin{tikzpicture}[node distance=1.5cm, every node/.style={circle, draw, minimum size=0.5cm}]
            \node (1) at (0,0) {A};
            \node (2) [above right of=1] {B};
            \node (3) [below right of=2] {C};
            \node (4) [below right of=1] {D};
            
            \path[->, thick, blue]
                (1) edge (2)
                (2) edge (3)
                (3) edge (4)
                (4) edge (1);
        \end{tikzpicture}
        \end{center}
        
        \column{0.6\textwidth}
        \small
        \textbf{Jak działa algorytm TSP?} \\
        Kiedy dron otrzymuje wiele punktów do zbadania, szuka najkrótszej drogi odwiedzającej każdy z nich.
        \begin{itemize}
            \item \textbf{Heurystyki (np. Najbliższy Sąsiad):} Zawsze leć do najbliższego, nieodwiedzonego punktu. Bardzo szybki, ale nie zawsze optymalny.
            \item \textbf{Algorytmy dokładne:} Przeliczają wszystkie ścieżki i szukają globalnego minimum. Stosuje się je, gdy punktów jest niewiele.
        \end{itemize}
    \end{columns}
\end{frame}

\begin{frame}{Zaawansowane misje ratunkowe 3D: Poligon}
    Gdy obszar do przeszukania ma nieregularny kształt (wielokąt), algorytm musi wyznaczyć równoległe ścieżki koszenia wewnątrz zadanego \textbf{poligonu}.
    
    \vspace{0.3cm}
    \textit{Kluczowe pojęcie: Pokrycie ścieżki i omijanie wypukłości (Ray-Casting Algorithm).}
    
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \small
            \textbf{Algorytm 'Lawnmower' (Koszenie trawnika)} \\
            Algorytm dzieli nieregularny wielokąt na serię równoległych linii. 
            Przecięcia tych linii z krawędziami wielokąta określają początek i koniec każdego przelotu (Ray-Casting). 
            Dron nawraca na krawędziach i kontynuuje skanowanie.
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{center}
            \begin{tikzpicture}[scale=0.8]
                % Poligon
                \draw[thick, fill=green!10] (0,0) -- (4,1) -- (5,4) -- (2,5) -- (-1,3) -- cycle;
                
                % Ścieżki koszenia
                \draw[->, red, thick] (-0.3, 0.5) -- (4.1, 0.5);
                \draw[->, red, thick] (4.1, 0.5) -- (4.3, 1.5) -- (-0.8, 1.5);
                \draw[->, red, thick] (-0.8, 1.5) -- (-0.9, 2.5) -- (4.5, 2.5);
                \draw[->, red, thick] (4.5, 2.5) -- (4.7, 3.5) -- (0, 3.5);
                \draw[->, red, thick] (0, 3.5) -- (1, 4.5) -- (2.5, 4.5);
                
                \node at (2, 2.5) [circle, fill=blue, inner sep=1.5pt] {};
                \node[blue, font=\tiny, above] at (2, 2.5) {Dron};
            \end{tikzpicture}
            \end{center}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Zaawansowane misje ratunkowe 3D: Cylinder}
    Zagrożenie może znajdować się w wysokim budynku lub w kominie. Tutaj dron skanuje powierzchnię walca lub jego objętość (np. ruchem po linii śrubowej / helisie).
    
    \vspace{0.3cm}
    \textit{Równanie helisy:} $x = r \cos(t), \quad y = r \sin(t), \quad z = c \cdot t$
    
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \small
            \textbf{Skanowanie 3D (Helisa)} \\
            Wykorzystywane np. do inspekcji turbin wiatrowych, wież lub kominów. 
            Wysokość $z$ rośnie jednostajnie z czasem, podczas gdy zmienne $x$ i $y$ kreślą okrąg wokół środka budynku.
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{center}
            \begin{tikzpicture}[scale=0.8]
                % Cylinder
                \draw[fill=blue!5, dashed] (0,0) ellipse (1.5 and 0.5);
                \draw[fill=blue!10] (-1.5,0) -- (-1.5,3) arc (180:360:1.5 and 0.5) -- (1.5,0) arc (360:180:1.5 and 0.5);
                \draw[fill=blue!20] (0,3) ellipse (1.5 and 0.5);
                
                % Helisa
                \draw[->, thick, red, domain=0:4*pi, samples=100, variable=\t] 
                    plot ({1.8*cos(\t r)}, {0.5*sin(\t r) + 0.2*\t});
                    
                \node at ({1.8*cos(3.5*pi r)}, {0.5*sin(3.5*pi r) + 0.2*3.5*pi}) [circle, fill=black, inner sep=1.5pt] {};
                \node[font=\tiny, right] at ({1.8*cos(3.5*pi r)}, {0.5*sin(3.5*pi r) + 0.2*3.5*pi}) {Dron};
            \end{tikzpicture}
            \end{center}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Zaawansowane misje ratunkowe 3D: Pościg}
    Co w przypadku, gdy cel (np. uciekający samochód) stale zmienia swoje położenie? Dron w czasie rzeczywistym musi aktualizować trajektorię, stosując \textbf{algorytmy pościgu} i szacując wektor prędkości celu.
    
    \vspace{0.3cm}
    \textit{Poprawka nawigacyjna (Proportional Navigation):} Przyspieszenie drona jest proporcjonalne do szybkości zmian kąta widzenia celu.
    
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \small
            \textbf{Nawigacja Proporcjonalna} \\
            Algorytm nie celuje tam, gdzie obiekt jest w danej chwili (bo by nie zdążył), ale wyprzedza cel.
            Kierunek i prędkość drona są korygowane proporcjonalnie do linii widzenia (Line of Sight) i przewidywanej ścieżki uciekającego.
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{center}
            \begin{tikzpicture}[scale=0.8]
                % Cel
                \node (target) at (4,2) [circle, fill=red, inner sep=2pt, label=above:Cel] {};
                \draw[->, dashed, red] (target) -- (5,2.5) node[right, font=\tiny] {$\vec{V}_{celu}$};
                
                % Dron
                \node (drone) at (0,0) [circle, fill=blue, inner sep=2pt, label=below:Dron] {};
                \draw[->, dashed, blue] (drone) -- (5,2.5) node[above left, font=\tiny] {$\vec{V}_{przewidywana}$};
                
                % Line of sight
                \draw[dotted, thick] (drone) -- (target) node[midway, sloped, above, font=\tiny] {Line of Sight};
            \end{tikzpicture}
            \end{center}
        \end{column}
    \end{columns}
\end{frame}
